async function fetchRealCandidateCities(candObj){ const state = $('state').value; const office = $('office').value; const round = $('round').value; if (state === 'br') { $('rightPanelTitle').textContent = `Cidades + Votadas: ${candObj.name}`; $('rightPanelSub').textContent = `Selecione um estado específico`; $('rows').innerHTML = `Selecione um estado nos botões acima para ver o ranking municipal.`; return; } $('rightPanelTitle').textContent = `Cidades + Votadas: ${candObj.name}`; $('rightPanelSub').textContent = `A consultar votação real por município...`; $('col1Title').textContent = 'Município'; $('rows').innerHTML = `Buscando dados oficiais do TSE...`; try { const r = await fetch(`/api/candidate-cities?number=${candObj.number}&state=${state}&office=${office}&round=${round}`); const d = await r.json(); if (d.ok && d.cities && d.cities.length > 0) { $('rightPanelSub').textContent = `${d.cities.length} municípios ordenados da maior para a menor votação`; $('rows').innerHTML = d.cities.map((c, idx) => `${idx + 1}º ${escapeHtml(c.cityName)}${fmt(c.votes)}${pct(c.percent)}`).join(''); } else { $('rows').innerHTML = `Nenhum voto computado nos municípios consultados para este candidato.`; } } catch (e) { $('rows').innerHTML = `Erro ao carregar os dados municipais.`; } }
